\title{Elementary inverse energy for aligned fibres}
\date{}

\begin{document}
\maketitle

\begin{abstract}
The remaining inverse step in \texttt{energy-increment.tex} is a clustering problem for a family of subsets, not a Freiman inverse-sumset problem.
A single common core is not always forced (two equal clusters of size $q/8$ already exceed twice-random energy).
What \emph{is} forced, when every fibre has size at least $q/3$, is a star of $\Omega(|T_x|)$ fibres that are pairwise $1/2$-close to a common $U$.
Independence then makes $U$ miss a positive fraction of $\Omega(d)$ of its own translates.
Half-intersection only gives $|U\cap(U+\sigma)|\le|U|/2$, which does not beat
the random prediction on non-heavy cores and does not feed the horizontal-packing Weyl estimate.
Exact cylinders, including AP-support cylinders of any common difference,
pack by the existing interval lemma after an affine change of
coordinates (Lean: \texttt{exact\_cylinder\_pack\_ap}), so they do not need Weyl.
Three-quarters of maximal energy produces a $\tfrac12$-close star at any fibre
size (Lean: \texttt{three\_quarters\_energy\_core}); a thin such star still does
not pack elementarily.
The leftover is intermediate energy on fibres smaller than $q/3$, together with
those thin three-quarters-energy stars that are not exact cylinders.
\end{abstract}

\section{Setup}

As in \texttt{energy-increment.tex}: $A\subset\F_q^{2}$ is $G_R$-independent, $B_x$ is the fibre over $x$, $A_t=B_{x+t}-t^{2}$ for $t\in T_x$, $W_x=\sum_t|A_t|$, and
\[
  E_x=\sum_{t,t'\in T_x}|A_t\cap A_{t'}|=\sum_{y\in\F_q}\mu(y)^{2},
\]
where $\mu(y)=\#\{t\in T_x:y\in A_t\}$.
Cauchy--Schwarz is $W_x^{2}\le|\cup A_t|\,E_x$.
The fibre is \emph{strictly mixing} if $E_x\le W_x^{2}/q$; then $|\cup A_t|\ge q$ and $B_x=\emptyset$ as soon as $W_x\ge q$ (Lean: \texttt{strict\_mixing\_empty}).

Independence between $x+s$ and $x+t$ is the disjointness
\begin{equation}\label{eq:shift}
  A_s\;\cap\;\bigl(A_t+2t(t-s)\bigr)\;=\;\emptyset
  \qquad(s-t\in T),
\end{equation}
proved in Lean as \texttt{shifted\_independent}.

\section{A single core is not forced}

\begin{lemma}[Two clusters exceed twice-random]\label{lem:twoclust}
Let $T_x=T_1\sqcup T_2$ with $|T_1|=|T_2|=s$, and let $A_t=U$ for $t\in T_1$, $A_t=V$ for $t\in T_2$, with $U\cap V=\emptyset$ and $|U|=|V|=q/8$.
Then $E_x=s^{2}q/4$ and $2W_x^{2}/q=s^{2}q/8$, so $E_x>2W_x^{2}/q$.
\end{lemma}

Thus Conjecture~12 of \texttt{energy-increment.tex}, as stated (``a single common $U$''), is false.
The right statement is a \emph{bounded} collection of cores.

\section{Medium fibres cluster}

Write $m_t=|A_t|$ and $\overline m=W_x/|T_x|$.
Twice-random energy means the average pairwise intersection is $E_x/|T_x|^{2}>2\overline m^{2}/q$.

\begin{lemma}[Many heavy pairs]\label{lem:heavypairs}
Suppose $\min_t m_t\ge q/3$ and $E_x>2W_x^{2}/q$.
Let $G$ be the graph on $T_x$ with an edge $t\sim t'$ whenever $|A_t\cap A_{t'}|\ge \overline m/2$.
Then $G$ has at least $|T_x|^{2}/9$ edges, hence a vertex of degree at least $|T_x|/9$.
\end{lemma}

\begin{proof}
The average intersection is $>2\overline m^{2}/q\ge 2\overline m/3$ because $\overline m\ge q/3$.
Pairs that are not edges contribute at most $(\overline m/2)|T_x|^{2}$ to $E_x$.
The remaining energy is at least
\[
  \bigl(2\overline m^{2}/q-\overline m/2\bigr)|T_x|^{2}
  \ge
  \overline m\bigl(2/3-1/2\bigr)|T_x|^{2}
  =
  \overline m\,|T_x|^{2}/6.
\]
Each edge contributes at most $\max m_t\le q/2$ (heavy fibres were already deleted), so the number of edges is at least
\[
  \frac{\overline m\,|T_x|^{2}}{6\cdot q/2}
  \ge
  \frac{(q/3)|T_x|^{2}}{3q}
  =
  \frac{|T_x|^{2}}{9}.
\]
\end{proof}

\begin{lemma}[Star is an approximate cylinder]\label{lem:star}
Let $t_0$ be a vertex of degree $\ge|T_x|/9$ in the graph of Lemma~\ref{lem:heavypairs}, and set $U=A_{t_0}$.
Then at least $|T_x|/9$ indices $t$ satisfy $|A_t\cap U|\ge\overline m/2$.
In particular, if all fibres have size in $[q/3,q/2]$, each such $A_t$ is $1/2$-close to $U$ in symmetric difference relative to $q$.
\end{lemma}

\section{Approximate cylinders are small}

\begin{lemma}[Half-intersection]\label{lem:half}
Let $U,A,B\subset\F_q$ satisfy $|A\cap U|\ge\tfrac34|U|$, $|B\cap U|\ge\tfrac34|U|$, and $A\cap(B+\sigma)=\emptyset$.
Then $|U\cap(U+\sigma)|\le|U|/2$.
\end{lemma}

\begin{proof}
Write $V=U\cap(U+\sigma)$ and $B_{\sigma}=B+\sigma$.
Then $A\cap V$ and $B_{\sigma}\cap V$ are disjoint subsets of $V$, so $|A\cap V|+|B_{\sigma}\cap V|\le|V|$.
Inclusion-exclusion inside $U$ gives $|A\cap V|\ge|A\cap U|+|V|-|U|$, and likewise $|B_{\sigma}\cap V|\ge|B\cap U|+|V|-|U|$.
Adding and rearranging: $|A\cap U|+|B\cap U|+|V|\le 2|U|$.
The density hypotheses give $|A\cap U|+|B\cap U|\ge\tfrac32|U|$, hence $|V|\le|U|/2$.
\end{proof}

On a $3/4$-dense star the shift~\eqref{eq:shift} and Lemma~\ref{lem:half} produce $|U\cap(U+\sigma)|\le|U|/2$ for every $\sigma=2t(t-s)$ with $s,t$ in the star and $s-t\in T$.
Lemma~\ref{lem:half} is Lean \texttt{half\_intersection}.

That overlap bound does \emph{not} feed the Weyl estimate used for horizontal packings.
Those packings need a set that \emph{misses} a rich difference set
($|Z\cap(Z+\delta)|=0$ for $\delta\in\Delta$), not one that meets it in half its points.
If $|U|\le q/2$, the random prediction is already
$|U\cap(U+\sigma)|\sim|U|^{2}/q\le|U|/2$
(Lean: \texttt{random\_overlap\_le\_half}).
Half-intersection is therefore vacuous on every non-heavy core, including
medium fibres of size $q/3$.
The earlier claim that Lemma~\ref{lem:half} plus Weyl gives $|U|=O(\log q)$
is false.

\begin{lemma}[Above twice-random beats the random overlap]\label{lem:Kstar}
Suppose every fibre at $x$ has size in $[q/3,q/2]$ and
$E_x\ge K W_x^{2}/q$ with $K>2$.
Let $G$ be the graph on $T_x$ with an edge whenever
$|A_t\cap A_{t'}|\ge\theta\overline m$, where
$\theta=(K+2)/6\in(2/3,\,K/3)$.
Then $G$ has at least $(K-2)|T_x|^{2}/9$ edges, hence a vertex of degree
$\Omega((K-2)|T_x|)$.
For each such edge, Lemma~\ref{lem:close} gives
$|A_t\cap(A_t+\sigma)|\le(1-\theta)|A_t|<|A_t|^{2}/q$.
\end{lemma}

\begin{proof}
The average intersection is at least $K\overline m^{2}/q\ge(K/3)\overline m$.
Non-edges contribute at most $\theta\overline m\,|T_x|^{2}$.
The remaining energy is at least
$(K/3-\theta)\overline m\,|T_x|^{2}=(K-2)\overline m\,|T_x|^{2}/6$.
Each edge contributes at most $q/2$, and $\overline m\ge q/3$, so the number
of edges is at least
\[
  \frac{(K-2)\overline m\,|T_x|^{2}}{6\cdot q/2}
  \ge
  \frac{(K-2)|T_x|^{2}}{9}.
\]
Comparable sizes give $|A_t\setminus A_{t'}|\le(1-\theta)|A_t|$.
The threshold $\theta>2/3$ is exactly the condition
$(1-\theta)|A_t|<|A_t|^{2}/q$ at density $1/3$.
\end{proof}

\begin{theorem}[Medium aligned fibres, corrected]\label{thm:medium}
If every fibre at $x$ has size in $[q/3,q/2]$ and $E_x>2W_x^{2}/q$, then
Lemma~\ref{lem:star} produces a $\tfrac12$-close star.
That star is a theorem; its size is not.
A size bound $O(q\log q)$ would need all three of:
\begin{enumerate}
  \item energy bounded away from twice-random ($K\ge 2+\eta$), so that
        Lemma~\ref{lem:Kstar} supplies a genuine discrepancy
        $|U\cap(U+\sigma)|<|U|^{2}/q$;
  \item a star neighbourhood that is dense in an affine interval of length
        $\Theta(d)$ (otherwise the shifts are an AP-poor set, the same
        leftover as two-clusters);
  \item the Weyl estimate of Proposition~4 in \texttt{structured-cases.tex},
        now applied to a set that is actually below the random overlap.
\end{enumerate}
At $K\downarrow 2$ the discrepancy is $o(1)$ and (1) fails.
We do \emph{not} claim that every medium aligned piece has size $O(q\log q)$.
\end{theorem}

\section{Exact cylinders pack without Weyl}

If $A_t=U$ for every $t\in T_x$, independence is complete disjointness
$U\cap(U+2t(t-s))=\emptyset$ (Lean: \texttt{exact\_cylinder\_shift}).
Theorem~\ref{thm:exactpack} (Lean: \texttt{exact\_cylinder\_pack}) then gives
$|U|\le q/(d'+1)=O(\sqrt{\log q})$ when the support contains $\{t,t+1,\dots,t+d'\}$.
The same packing holds for any common difference (Theorem~\ref{thm:exactpackap}).
The cylinder occupies $O(d)$ $x$-coordinates, so the piece has size $O(q)$.
Weyl is not used.

The two-cluster example of Lemma~\ref{lem:twoclust} therefore packs as soon as
each colour class contains a long arithmetic progression:
each block is an exact AP-cylinder of size $O(q)$.
A leftover two-cluster must have AP-poor supports in $T$
(Behrend-type subsets of $\{1,\dots,d\}$ are large enough to keep
$E_x$ above twice-random).
We do not claim those AP-poor clusters are packed.

\section{Three-quarters energy, any fibre size}

Maximal energy is $|T_x|W_x$, because $\mu\le|T_x|$ pointwise
(Lean: \texttt{energy\_le\_card\_mul\_mass}).
The next two lemmas need no lower bound on $|A_t|$.

\begin{lemma}[Three-quarters energy core]\label{lem:34core}
If $E_x\ge\tfrac34|T_x|W_x$ and $W_x>0$, write
$U=\{y:\mu(y)\ge|T_x|/2\}$.
Then $\sum_{y\in U}\mu(y)^{2}\ge E_x/4$.
\end{lemma}

\begin{proof}
On the complement one has $2\mu<|T_x|$, hence
$2\sum_{U^{c}}\mu^{2}\le(|T_x|-1)W_x$.
The hypothesis $4E_x\ge 3|T_x|W_x$ rearranges to
$3E_x\ge 2(|T_x|-1)W_x$, so
$4\sum_{U^{c}}\mu^{2}\le 3E_x$ and therefore
$E_x\le 4\sum_U\mu^{2}$.
This is Lean \texttt{three\_quarters\_energy\_core}.
\end{proof}

\begin{lemma}[Three-quarters energy star]\label{lem:34star}
Suppose $E_x\ge\tfrac34|T_x|W_x$ and $|T_x|\ge 2$.
Let $G$ be the graph on $T_x$ with an edge $t\sim t'$ whenever
$|A_t\cap A_{t'}|\ge\tfrac12\min(m_t,m_{t'})$.
Then $G$ has $\Omega(|T_x|^{2})$ edges, hence a vertex of degree $\Omega(|T_x|)$.
\end{lemma}

\begin{proof}
The average pairwise intersection is $E_x/|T_x|^{2}\ge\tfrac34\,\overline m$.
Pairs that are not edges contribute at most
$\tfrac12\overline m\,|T_x|^{2}$ if one uses
$\min(m_t,m_{t'})\le\max(m_t,m_{t'})$ and compares to the mean
(or, after discarding an $O(1)$-fraction of fibres whose size is
far from $\overline m$, the remaining pairs have
$\min(m_t,m_{t'})\ge\tfrac12\overline m$).
The leftover energy is then at least
$\bigl(\tfrac34-\tfrac12\bigr)\overline m\,|T_x|^{2}=\overline m\,|T_x|^{2}/4$,
so there are $\Omega(|T_x|^{2})$ edges.
\end{proof}

Unlike Lemma~\ref{lem:heavypairs}, this uses energy close to the
\emph{maximum}, not merely twice-random, so the density cutoff $\delta\ge 1/3$
is not needed.
A fibre of any size with three-quarters-maximal energy still produces an
$\Omega(|T_x|)$-star of pairwise $\tfrac12$-close sets.

\begin{lemma}[Close fibres almost miss their translate]\label{lem:close}
If $A\cap(B+\sigma)=\emptyset$, then $|A\cap(A+\sigma)|\le|A\setminus B|$
(Lean: \texttt{close\_sets\_almost\_disjoint\_translates}).
In particular, if $A$ and $B$ are $\varepsilon$-close of equal size, $|A\cap(A+\sigma)|\le\varepsilon|A|$.
\end{lemma}

\begin{theorem}[Three-quarters energy is structured, not packed]\label{thm:nearmax}
Suppose $E_x\ge\tfrac34|T_x|W_x$ and $|T_x|\ge 2$.
Then a star of $\Omega(|T_x|)$ fibres is pairwise $\tfrac12$-close to a common $U$
(Lemmas~\ref{lem:34core}--\ref{lem:34star}).
\begin{itemize}
  \item If $|U|\ge q/3$, the star is a medium approximate cylinder:
        Lemma~\ref{lem:star} applies, but Theorem~\ref{thm:medium} is not a
        size bound.
  \item If the fibres are exactly equal to $U$ along an interval of $T$-steps,
        Theorem~\ref{thm:exactpack} below gives $|U|\le q/(d'+1)$ with no Weyl.
  \item If $|U|=o(q)$ and the fibres are only $\tfrac12$-close, Lemma~\ref{lem:close}
        yields $|U\cap(U+\sigma)|\le|U|/2$.
        That is the same vacuous half-intersection as in Section~7:
        a single $\varepsilon$-fringe cannot be deleted for $\Theta(d)$ many
        shifts $\sigma$ unless $\varepsilon\ll 1/d$, i.e.\ unless the fibres
        are essentially exact.
        Thin three-quarters-energy stars remain in the leftover.
\end{itemize}
\end{theorem}

The earlier claim that an $\varepsilon$-fringe reduces near-maximal energy to
the interval lemma was too strong:
$\bigcup_{\sigma}(U\cap(U+\sigma))$ can cover $U$ as soon as
$d'\varepsilon\ge 1$.
Elementary packing is only for exact cylinders.

\begin{theorem}[Exact cylinder packing]\label{thm:exactpack}
If $A_{t+k}=U$ for every $k\in\{0,\dots,d'\}\subset T\cup\{0\}$, $t\ne 0$,
and $q$ is an odd prime, then $|U|\cdot(d'+1)\le q$.
\end{theorem}

\begin{proof}
Independence is $U\cap(U-2tk)=\emptyset$ for each $k=1,\dots,d'$
(Lean: \texttt{exact\_cylinder\_shift}).
The forbidden differences are the affine image $(-2t)\cdot\{1,\dots,d'\}$
with $-2t\ne 0$.
Lean \texttt{exact\_cylinder\_pack} applies \texttt{affine\_diffFree\_card\_le}.
\end{proof}

\begin{theorem}[AP-support exact cylinders]\label{thm:exactpackap}
If $A_{t+ak}=U$ for every $k\in\{0,\dots,d'\}$, $t\ne 0$, $a\ne 0$,
$\{a,2a,\dots,d'a\}\subset T$, and $q$ is an odd prime, then
$|U|\cdot(d'+1)\le q$.
\end{theorem}

\begin{proof}
The forbidden differences are $(-2ta)\cdot\{1,\dots,d'\}$ with $-2ta\ne 0$.
Lean \texttt{exact\_cylinder\_pack\_ap}.
The piece over the progression has size at most $q$.
\end{proof}

The high-multiplicity core of a three-quarters-energy fibre has size
$O(\overline m)$:
$|U|\cdot|T_x|^{2}\le 4E_x\le 4|T_x|W_x$, hence $|U|\le 4\overline m$
(Lean: \texttt{high\_multiplicity\_core\_card}).
A thin fibre cannot hide a large core.

\section{Why thin cores are a different problem}

Lemma~\ref{lem:half} concludes $|U\cap(U+\sigma)|\le|U|/2$.
A random $\delta$-dense set already has $|U\cap(U+\sigma)|\sim\delta|U|$, which is at most $|U|/2$ as soon as $\delta\le 1/2$.
Half-intersection is therefore vacuous for every non-heavy core --- including
medium fibres of size $q/3$, not only vanishing-density cores.

The intersection-graph argument is also density-limited.
If $\min_t m_t\ge\delta q$ and $E_x>2W_x^{2}/q$, the average pairwise intersection is $>2\delta\,\overline m$.
An edge threshold $\theta\overline m$ with $\theta<2\delta$ produces $\Omega(\delta^{2}|T_x|^{2})$ edges, hence a vertex of degree $\Omega(\delta^{2}|T_x|)$.
Constant-fraction overlap ($\theta=\Theta(1)$) requires $\delta>\Theta(1)$; the $q/3$ cutoff in Lemma~\ref{lem:heavypairs} is essentially optimal for twice-random energy.
Degree at least $1$ requires $\delta\gg 1/\sqrt{d}$, i.e.\ fibre size $\gg\sqrt{q\log q}$.
Below that, twice-random energy need not produce even a single heavy pair.

\section{Popular cores for smaller fibres}

Define $U_{\mathrm{pop}}=\{y:\mu(y)\ge E_x/(2W_x)\}$.
On the complement one has $\mu<E_x/(2W_x)$, hence $\sum_{U_{\mathrm{pop}}^{c}}\mu^{2}<E_x/2$: the popular core carries at least half the energy
(Lean: \texttt{popular\_core\_energy}).
Peeling $U_{\mathrm{pop}}$ and repeating produces $O(\log q)$ cores plus a strictly mixing fringe.

Each core of \emph{constant} density produces a close star
(Lemma~\ref{lem:star} or Lemma~\ref{lem:Kstar}); packing that star is
Theorem~\ref{thm:medium}, which is not a size bound.
A core of vanishing density $\delta=o(1)$ is constrained by the interval lemma only when the fibres are exactly equal to it (Theorem~\ref{thm:exactpack}).
Three-quarters energy still produces a $\tfrac12$-close star at any size (Theorem~\ref{thm:nearmax}), but a thin $\tfrac12$-close star does not pack elementarily.
The remaining window is intermediate energy
\[
  2W_x^{2}/q\;<\;E_x\;<\;\tfrac34|T_x|W_x
\]
on fibres of size in $\bigl(q^{1/3}(\log q)^{-O(1)},\,q/3\bigr)$,
together with three-quarters-energy stars whose core is still that thin,
medium $\tfrac12$-close stars that do not beat the random overlap,
and AP-poor stars.

Fibres of size $\le q^{1/3}/(\log q)^{10/3}$ contribute at most $q^{4/3}/(\log q)^{10/3}$ points.
After the lift this is $O\bigl(q^{4/3}(\log q)^{-4/3}\bigr)=o(n^{2/3})$, so those fibres cannot prevent an Alon-beating bound.
They can still prevent a SOTA bound.

\section{What this changes}

\begin{theorem}[Updated dichotomy]\label{thm:dichotomy2}
Let $A$ be $G_R$-independent and $q>3$ large.
Then $A$ is a union of
\begin{enumerate}
  \item a heavy-fibre piece of size $O(q\sqrt{\log q})$;
  \item a $T$-sparse piece of size $O(q\sqrt{\log q})$;
  \item a strictly-mixing $T$-dense piece of size $O(q\sqrt{\log q})$;
  \item exact cylinders, including those whose support is an arithmetic
        progression of any common difference, each of size $O(q)$
        (interval packing; Lean \texttt{exact\_cylinder\_pack\_ap}; no Weyl);
  \item medium ($|B_x|\ge q/3$) twice-random fibres have a $\tfrac12$-close
        star (Lemma~\ref{lem:star}); a $(2/3+)$ star with overlap below
        $|U|^{2}/q$ exists when $K$ is bounded away from $2$
        (Lemma~\ref{lem:Kstar});
  \item a leftover consisting of intermediate-energy fibres of size in
        $\bigl(q^{1/3}(\log q)^{-O(1)},\,q/3\bigr)$,
        thin three-quarters-energy stars that are not exact cylinders,
        two-cluster and medium stars whose supports in $T$ are AP-poor,
        and medium stars whose energy is only barely above twice-random.
\end{enumerate}
Items (1)--(4) are theorems (no Weyl).
Item (5) is a structure theorem, not a size bound.
Item (6) is the unwritten inverse step, and now also includes the medium
pieces previously recorded as packed.
\end{theorem}

We still do \emph{not} claim $\alpha(G_R)=O(q\log q)$ as a finished theorem.
We do claim that the inverse-energy gap is no longer a Freiman loss $q^{\delta}$,
that exact cylinders pack elementarily, that three-quarters energy produces a
star at any fibre size, and that half-intersection does not pack medium cores.
